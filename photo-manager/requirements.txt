# MMR Photo Manager — Python dependencies
# Install with: pip install -r requirements.txt
#
# Minimum versions are tested working versions.
# Pin exact versions with == for reproducible builds (CI / team environments).

# ── Core (required for all modules) ──────────────────────────────
opencv-python>=4.8,<5.0       # image loading, Laplacian, HOG fallback
Pillow>=10.0                  # image I/O, JPEG draft() fast-load
numpy>=1.24                   # array operations
tqdm>=4.66                    # progress bar

# ── HEIC / iPhone photo support ──────────────────────────────────
pillow-heif>=0.13             # HEIC/HEIF format (iPhone Live Photos)

# ── Person detection ─────────────────────────────────────────────
# Downloads yolov8n.pt (~6 MB) on first use. Requires internet on first run.
# Falls back to OpenCV HOG detector if model cannot be downloaded.
ultralytics>=8.0              # YOLOv8 person detection

# ── Bib number OCR ───────────────────────────────────────────────
# Downloads OCR models (~100 MB) on first use. Requires internet on first run.
# Falls back to pytesseract (see below) if models cannot be downloaded.
easyocr>=1.7                  # primary OCR engine (better accuracy)

# pytesseract is a Python wrapper only — Tesseract binary must be installed
# separately on the OS:
#   macOS:  brew install tesseract
#   Ubuntu: sudo apt install tesseract-ocr
#   Windows: https://github.com/UB-Mannheim/tesseract/wiki
pytesseract>=0.3.10           # fallback OCR engine (offline-capable)

# ── Posture estimation (optional, improves accuracy) ─────────────
# Without mediapipe, posture falls back to bounding-box heuristics.
# mediapipe does not yet support Python 3.12+ on Apple Silicon — skip if install fails.
# mediapipe>=0.10             # pose keypoints for running/standing/bent detection

# ── Phase 3: Facial recognition ──────────────────────────────────

# Option A — Azure Face API (cloud, recommended for production)
# Requires AZURE_FACE_KEY + AZURE_FACE_ENDPOINT in .env.local
# Setup: copy .env.local.example → .env.local and fill in your Azure keys
azure-ai-vision-face>=1.0.0b2 # Azure Face API: detect, verify, identify
python-dotenv>=1.0            # loads .env.local automatically

# Option B — local face_recognition (offline, used by bib_analyzer.py)
# face_recognition requires cmake + dlib to build from source.
#
# macOS install:
#   brew install cmake
#   pip install dlib
#   pip install face_recognition
#
# Full guide: https://github.com/ageitgey/face_recognition#installation
face_recognition>=1.3         # face embedding + member photo matching (local fallback)
