  FROM v_last_successful_batch;

Failed batches (for retry logic):

  SELECT JobID, ConfigKey, BatchNumber, ErrorMessage
  FROM sheets_sync_log
  WHERE Status = 'error'
  ORDER BY StartedAt DESC;
